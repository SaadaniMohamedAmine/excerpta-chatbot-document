import { Request, Response, NextFunction } from 'express';
import jwt from 'jsonwebtoken';
interface AuthRequest extends Request {
user?: any;
}
export const requireAuth = (req: AuthRequest, res: Response, next: NextFunction) => {
const authHeader = req.headers.authorization;
if (!authHeader) {
return res.status(401).json({ error: 'Authorization header missing' });
}
const parts = authHeader.split(' ');
if (parts.length !== 2 || parts[0] !== 'Bearer') {
return res.status(401).json({ error: 'Authorization header format must be Bearer [token]' });
}
const token = parts[1];
try {
const payload = jwt.verify(token, process.env.JWT_SECRET!);
req.user = payload;
next();
} catch (err) {
return res.status(401).json({ error: 'Invalid or expired token' });
}
};

How is the user identity attached to the request in the authentication middleware?

auto_awesome Excerpta Assistant

The authentication middleware verifies the JSON Web Token (JWT) provided in the Authorization header. Upon successful verification, the decoded payload, which contains the user's identity claims, is explicitly attached to the Express request object.

Specifically, the payload is assigned to the req.user property before passing control to the next middleware via next() link [auth_middleware.ts, l. 24] . This allows subsequent route handlers to access the authenticated user's details directly from the request context.
